# --- Security by default ---
# Argon2id password hashing + AES-256-GCM encryption at rest (KER_DATA_KEY).
# pyproject.toml already declares these as project dependencies, but the
# Docker image installs from THIS file and then runs `pip install --no-deps .`
# — so pyproject's dependency list is never actually consulted at build time.
# These two lines are what makes the built image match what pyproject.toml
# promises. Without them: password hashing silently falls back to scrypt,
# and if KER_DATA_KEY is ever set, jarvis/security/crypto.py fails loud on
# startup (RuntimeError) instead of storing data unencrypted.
argon2-cffi>=23.1,<26.0
cryptography>=41.0
